cmake_minimum_required(VERSION 3.22)
project(turnip_test C)
find_package(Vulkan QUIET)
add_executable(test_turnip test_turnip.c)
target_link_libraries(test_turnip dl)
if(Vulkan_FOUND)
  target_link_libraries(test_turnip Vulkan::Vulkan)
  target_include_directories(test_turnip PRIVATE ${Vulkan_INCLUDE_DIRS})
endif()
# fall back to NDK vulkan headers if no package
if(NOT Vulkan_FOUND)
  if(DEFINED ENV{ANDROID_NDK_ROOT})
    target_include_directories(test_turnip PRIVATE $ENV{ANDROID_NDK_ROOT}/toolchains/llvm/prebuilt/linux-x86_64/sysroot/usr/include)
  endif()
endif()
